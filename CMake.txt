cmake_minimum_required(VERSION 3.16)
project(EPIC CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Try to find OpenMP
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
    message(STATUS "Found OpenMP: ${OpenMP_CXX_VERSION}")
else()
    message(STATUS "OpenMP not found, using manual detection")
    if(APPLE)
        set(OpenMP_CXX_FLAGS "-Xpreprocessor -fopenmp -I/opt/homebrew/opt/libomp/include")
        set(OpenMP_CXX_LIBRARIES "/opt/homebrew/opt/libomp/lib/libomp.dylib")
        set(OpenMP_CXX_VERSION "14.0.0")
    endif()
endif()

add_executable(env_two_particle_test
    tests/env_two_particle_test.cpp
)

add_executable(orbit_test
    tests/orbit_test.cpp
)

target_include_directories(env_two_particle_test PRIVATE ${CMAKE_SOURCE_DIR})
target_include_directories(orbit_test PRIVATE ${CMAKE_SOURCE_DIR})

if(OpenMP_CXX_FOUND)
    target_link_libraries(env_two_particle_test PRIVATE OpenMP::OpenMP_CXX)
    target_link_libraries(orbit_test PRIVATE OpenMP::OpenMP_CXX)
elseif(OpenMP_CXX_LIBRARIES)
    target_compile_options(env_two_particle_test PRIVATE ${OpenMP_CXX_FLAGS})
    target_compile_options(orbit_test PRIVATE ${OpenMP_CXX_FLAGS})
    target_link_libraries(env_two_particle_test PRIVATE "${OpenMP_CXX_LIBRARIES}")
    target_link_libraries(orbit_test PRIVATE "${OpenMP_CXX_LIBRARIES}")
endif()

target_compile_options(env_two_particle_test
    PRIVATE
        $<$<CONFIG:Release>: -DNDEBUG>
        $<$<CONFIG:RelWithDebInfo>: -DNDEBUG>
)
